IMG_SIZE  = (224, 224)  # EfficientNetB3 expects 224x224
BATCH     = 32          # 32 images per training step
NUM_CLASS = 10          # butterfly, cat, chicken, cow, dog,
                        # elephant, horse, sheep, spider, squirrel

TRAIN = "/content/dataset/train"
VAL   = "/content/dataset/validation"
TEST  = "/content/dataset/test"

# Training generator: normalise + augment
# Augmentation = show the model flipped/rotated versions
# so it learns to recognise animals from any angle
train_gen = ImageDataGenerator(
    preprocessing_function=preprocess_input,
    horizontal_flip=True,       # mirror image left-right
    rotation_range=20,          # rotate up to 20 degrees
    zoom_range=0.15,            # zoom in or out slightly
    width_shift_range=0.1,      # shift image left/right
    height_shift_range=0.1,     # shift image up/down
    brightness_range=[0.8,1.2], # random brightness
    fill_mode="nearest"
)

# Val/test: normalise ONLY — never augment these
val_gen = ImageDataGenerator(preprocessing_function=preprocess_input)

train_data = train_gen.flow_from_directory(
    TRAIN, target_size=IMG_SIZE,
    batch_size=BATCH, class_mode="categorical"
)
val_data = val_gen.flow_from_directory(
    VAL, target_size=IMG_SIZE,
    batch_size=BATCH, class_mode="categorical"
)
test_data = val_gen.flow_from_directory(
    TEST, target_size=IMG_SIZE, batch_size=BATCH,
    class_mode="categorical", shuffle=False
    # shuffle=False is critical for confusion matrix
    # labels must stay in the same order as predictions
)

print("Classes:", train_data.class_indices)
print(f"Train: {train_data.samples} | Val: {val_data.samples} | Test: {test_data.samples}")

# Load EfficientNetB3 pretrained on ImageNet
# include_top=False removes its original 1000-class output
# We replace it with our own 10-class output
base = EfficientNetB3(
    weights="imagenet",
    include_top=False,
    input_shape=(224, 224, 3)
)

# FREEZE the base — don't change what EfficientNet knows
# training=False also keeps BatchNorm layers frozen
# (important — without this, features shift every batch)
base.trainable = False

# Build our full model
inputs  = tf.keras.Input(shape=(224, 224, 3))
x       = base(inputs, training=False)

# GlobalAveragePooling: reduces each feature map to 1 number
# Much better than Flatten — avoids 22M parameter Dense layer
x       = layers.GlobalAveragePooling2D()(x)
x       = layers.BatchNormalization()(x)
x       = layers.Dense(256, activation="relu")(x)
x       = layers.Dropout(0.4)(x) # randomly turn off 40% of neurons
                                  # during training → less overfitting
outputs = layers.Dense(NUM_CLASS, activation="softmax")(x)
# softmax: converts raw scores to probabilities that add to 1.0
# highest probability = predicted animal

model = models.Model(inputs, outputs)

model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-3),
    # Adam: smart optimizer that adjusts step size per weight
    # 1e-3 = 0.001 — good starting LR for head-only training
    loss="categorical_crossentropy",
    # standard loss for multi-class problems (more than 2 classes)
    metrics=["accuracy"]
)

model.summary() # check: trainable params should be ~600K not 22M

# Unfreeze the last 30 layers of EfficientNetB3
# Earlier layers (edges, lines, basic shapes) stay frozen
# Later layers (complex patterns) get fine-tuned on our animals
base.trainable = True
for layer in base.layers[:-30]:
    layer.trainable = False

# CRITICAL: recompile with a much smaller learning rate
# 1e-5 = 0.00001 — 100x smaller than Phase 1
# Large LR here would destroy EfficientNet's pretrained weights
model.compile(
    optimizer=tf.keras.optimizers.Adam(learning_rate=1e-5),
    loss="categorical_crossentropy",
    metrics=["accuracy"]
)

    img = image.load_img(img_path, target_size=(224, 224))
    img_array = image.img_to_array(img)
    img_batch = np.expand_dims(img_array, axis=0)

    print(f" Predicted     : {predicted_animal}")
    print(f" Confidence    : {confidence:.2f}%")
